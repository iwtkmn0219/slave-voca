```ts
export class NoTextError extends Error {
  constructor() {
    super('텍스트를 찾지 못했습니다. 스캔 PDF는 지원하지 않습니다.');
    this.name = 'NoTextError';
  }
}

const MIN_TEXT_CHARS = 200;
// 대소문자를 구분한다: 본문의 소문자 "references"가 줄 바꿈으로 혼자 남아도 제목으로 보지 않는다.
// 앞에 장 번호("6 References", "7. REFERENCES")가 붙어도 제목으로 본다.
const REFERENCES_HEADING =
  /^[ \t]*(?:\d+(?:\.\d+)*\.?[ \t]+)?(References|REFERENCES|Bibliography|BIBLIOGRAPHY|참고문헌)[ \t]*$/gm;
const LINE_END_HYPHEN = /([A-Za-z]+)-\n([a-z]+)/g;
const URL = /https?:\/\/\S+/g;
const EMAIL = /\S+@\S+/g;

export interface CleanOptions {
  // 빈도표에 있는 단어인지. 주면 줄 끝에서 잘린 합성어(fine-\ntuning)의 하이픈을 남긴다.
  isKnownWord?: (word: string) => boolean;
}

function restoreHyphen(left: string, right: string, isKnownWord?: (w: string) => boolean) {
  const joined = left + right;
  const lower = (w: string) => w.toLowerCase();
  const keepHyphen =
    isKnownWord !== undefined &&
    !isKnownWord(lower(joined)) &&
    isKnownWord(lower(left)) &&
    isKnownWord(lower(right));
  return keepHyphen ? `${left}-${right}` : joined;
}

// 합자(ﬁ 등) 펼치기 → 줄 끝 하이픈 복원 → 이메일·URL 제거 → 마지막 참고문헌 제목 이후 제거 → 공백 정리
export function cleanText(raw: string, options: CleanOptions = {}): string {
  let text = raw.normalize('NFKC');
  text = text.replace(LINE_END_HYPHEN, (_, left: string, right: string) =>
    restoreHyphen(left, right, options.isKnownWord),
  );
  text = text.replace(URL, ' ').replace(EMAIL, ' ');
  const heading = [...text.matchAll(REFERENCES_HEADING)].at(-1);
  if (heading) text = text.slice(0, heading.index);
  return text.replace(/\s+/g, ' ').trim();
}

export function splitSentences(text: string): string[] {
  const segmenter = new Intl.Segmenter('en', { granularity: 'sentence' });
  return [...segmenter.segment(text)]
    .map((s) => s.segment.trim())
    .filter((s) => s.length > 0);
}

export async function extractPdfText(bytes: Uint8Array): Promise<string> {
  const { getDocument } = await import('pdfjs-dist/legacy/build/pdf.mjs');
  // pdf.js는 넘겨받은 버퍼를 분리(detach)하므로 복사본을 넘긴다.
  const task = getDocument({ data: new Uint8Array(bytes), verbosity: 0 });
  try {
    const doc = await task.promise;
    const pages: string[] = [];
    for (let i = 1; i <= doc.numPages; i++) {
      const page = await doc.getPage(i);
      const content = await page.getTextContent();
      pages.push(
        content.items
          .map((item) => ('str' in item ? item.str + (item.hasEOL ? '\n' : '') : ''))
          .join(''),
      );
    }
    return pages.join('\n');
  } finally {
    await task.destroy();
  }
}

export async function parsePdf(bytes: Uint8Array, options: CleanOptions = {}): Promise<string[]> {
  const text = cleanText(await extractPdfText(bytes), options);
  if (text.replace(/\s/g, '').length < MIN_TEXT_CHARS) throw new NoTextError();
  return splitSentences(text);
}
```

```ts
import frequency from '../../../data/frequency.json';

export type RankOf = (word: string) => number | undefined;

const ranks = frequency as Record<string, number>;

// 빈도 순위(1 = 가장 흔함). 목록에 없으면 undefined.
// hasOwn: 'constructor' 같은 프로토타입 키를 순위로 오인하지 않는다.
export const rankOf: RankOf = (word) => (Object.hasOwn(ranks, word) ? ranks[word] : undefined);

export const isFrequentWord = (word: string): boolean => rankOf(word) !== undefined;
```

```ts
import { lemmatize } from './lemma';
import type { RankOf } from './frequency';

export interface Candidate {
  lemma: string;
  surface: string;
  sentence: string;
  count: number;
  isExisting: boolean;
  inDictionary: boolean;
}

export interface ExtractOptions {
  excludeTopN: number;
  rankOf: RankOf;
  knownLemmas: ReadonlySet<string>;
  existingLemmas: ReadonlySet<string>;
  isAcademic: (lemma: string) => boolean;
  inDictionary: (lemma: string) => boolean;
}

const TOKEN = /\p{L}+(?:-\p{L}+)*/gu;
const NON_ASCII = /[^\x00-\x7F]/;
const MIN_LENGTH = 3;
const MAX_SENTENCE_LENGTH = 300;
// 학술 핵심어 예외는 상위 2,000위 밖에만 적용한다. 그 안의 AWL 단어(area, data, code)는 쉽다.
const ACADEMIC_RESCUE_MIN_RANK = 2001;

interface Tally {
  lemma: string;
  surface: string;
  sentence: string;
  count: number;
  seenLowercase: boolean;
  seenCapitalizedMidSentence: boolean;
}

function bestRank(words: string[], rankOf: RankOf): number | undefined {
  const ranks = words.map(rankOf).filter((r): r is number => r !== undefined);
  return ranks.length > 0 ? Math.min(...ranks) : undefined;
}

function isExcludedByRank(lower: string, lemma: string, opts: ExtractOptions): boolean {
  // 표면형도 본다: data → datum처럼 원형이 드문 흔한 단어를 거른다.
  const rank = bestRank([lower, lemma], opts.rankOf);
  if (rank === undefined || rank > opts.excludeTopN) return false;
  return !(rank >= ACADEMIC_RESCUE_MIN_RANK && opts.isAcademic(lemma));
}

// 300자를 넘으면 단어가 보이도록 단어 주변을 잘라 정확히 300자로 맞춘다.
export function truncateAround(sentence: string, index: number, length: number): string {
  if (sentence.length <= MAX_SENTENCE_LENGTH) return sentence;
  const centered = index - Math.floor((MAX_SENTENCE_LENGTH - length) / 2);
  const start = Math.min(Math.max(0, centered), sentence.length - MAX_SENTENCE_LENGTH);
  const end = start + MAX_SENTENCE_LENGTH;
  const head = start > 0 ? '…' : '';
  const tail = end < sentence.length ? '…' : '';
  return head + sentence.slice(start + head.length, end - tail.length) + tail;
}

function compareByCountThenLemma(a: Candidate, b: Candidate): number {
  return b.count - a.count || (a.lemma < b.lemma ? -1 : a.lemma > b.lemma ? 1 : 0);
}

// 원형 처리기가 못 바꾼 복수형(datasets)을 같은 문서의 단수형 후보(dataset)에 합친다.
function mergePlurals(tallies: Tally[]): Tally[] {
  const byLemma = new Map(tallies.map((t) => [t.lemma, t]));
  const merged: Tally[] = [];
  for (const t of tallies) {
    const singular =
      t.lemma.endsWith('s') && !t.lemma.endsWith('ss') ? byLemma.get(t.lemma.slice(0, -1)) : undefined;
    if (singular) singular.count += t.count;
    else merged.push(t);
  }
  return merged;
}

export function extractCandidates(sentences: string[], opts: ExtractOptions): Candidate[] {
  const tallies = new Map<string, Tally>();

  for (const sentence of sentences) {
    let position = 0;
    for (const match of sentence.matchAll(TOKEN)) {
      const token = match[0];
      const isFirst = position++ === 0;
      if (token.includes('-') || token.length < MIN_LENGTH) continue;
      if (NON_ASCII.test(token)) continue; // Łukasz → ukasz 같은 조각 방지
      if (token === token.toUpperCase()) continue; // 약어: CNN, BERT
      const lower = token.toLowerCase();
      const lemma = lemmatize(lower);
      if (isExcludedByRank(lower, lemma, opts) || opts.knownLemmas.has(lemma)) continue;

      let tally = tallies.get(lemma);
      if (!tally) {
        tally = {
          lemma,
          surface: lower,
          sentence: truncateAround(sentence, match.index, token.length),
          count: 0,
          seenLowercase: false,
          seenCapitalizedMidSentence: false,
        };
        tallies.set(lemma, tally);
      }
      tally.count++;
      if (token === lower) tally.seenLowercase = true;
      else if (!isFirst) tally.seenCapitalizedMidSentence = true;
    }
  }

  const kept = [...tallies.values()].filter(
    (t) => t.seenLowercase || !t.seenCapitalizedMidSentence, // 고유명사 제외
  );
  return mergePlurals(kept)
    .map((t) => ({
      lemma: t.lemma,
      surface: t.surface,
      sentence: t.sentence,
      count: t.count,
      isExisting: opts.existingLemmas.has(t.lemma),
      inDictionary: opts.inDictionary(t.lemma),
    }))
    .sort(compareByCountThenLemma);
}
```

```ts
import { isAcademic } from '@/lib/extract/academic';
import { inDictionary } from '@/lib/extract/dictionary';
import { extractCandidates } from '@/lib/extract/extract';
import { isFrequentWord, rankOf } from '@/lib/extract/frequency';
import { NoTextError, parsePdf } from '@/lib/pdf/parse';
import { MAX_PDF_BYTES, pdfSizeError } from '@/lib/upload/limits';
import type { Db } from '../db/client';
import type { DocumentKind, DocumentStatus } from '../db/schema';
import { replaceCandidates } from '../repo/candidates';
import { createDocument, updateDocumentStatus } from '../repo/documents';
import { getSettings } from '../repo/settings';
import { listKnownLemmas, listWordLemmas } from '../repo/vocabulary';

export { MAX_PDF_BYTES };

export class UploadError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'UploadError';
  }
}

export interface IngestInput {
  fileName: string;
  title: string;
  kind: DocumentKind;
  bytes: Uint8Array;
}

export interface IngestResult {
  documentId: number;
  status: DocumentStatus;
}

const PDF_MAGIC = '%PDF-';
// PDF 규격은 파일 앞 1,024바이트 안에 헤더가 있으면 허용한다(앞에 붙은 쓰레기 바이트).
const PDF_MAGIC_WINDOW = 1024;

export function validateUpload(bytes: Uint8Array): void {
  const sizeError = pdfSizeError(bytes.length);
  if (sizeError) throw new UploadError(sizeError);
  const head = new TextDecoder('latin1').decode(bytes.subarray(0, PDF_MAGIC_WINDOW));
  if (!head.includes(PDF_MAGIC)) throw new UploadError('PDF 파일이 아닙니다.');
}

export function defaultTitle(fileName: string): string {
  return fileName.replace(/\.pdf$/i, '').trim() || '제목 없음';
}

// 업로드 검증 → 문서 생성 → 문장 추출 → 후보 저장. 처리 실패는 문서 상태(error)로 남긴다.
export async function ingestPdf(db: Db, input: IngestInput): Promise<IngestResult> {
  validateUpload(input.bytes);
  const title = input.title.trim() || defaultTitle(input.fileName);
  const doc = createDocument(db, { title, kind: input.kind });

  try {
    const sentences = await parsePdf(input.bytes, { isKnownWord: isFrequentWord });
    const settings = getSettings(db);
    const found = extractCandidates(sentences, {
      excludeTopN: settings.excludeTopN,
      rankOf,
      knownLemmas: listKnownLemmas(db),
      existingLemmas: listWordLemmas(db),
      isAcademic,
      inDictionary,
    });
    replaceCandidates(db, doc.id, found);
    const status: DocumentStatus = found.length === 0 ? 'done' : 'reviewing';
    updateDocumentStatus(db, doc.id, status);
    return { documentId: doc.id, status };
  } catch (err) {
    if (!(err instanceof NoTextError)) console.error('PDF 처리 실패', err);
    const message = err instanceof NoTextError ? err.message : 'PDF를 처리하지 못했습니다.';
    updateDocumentStatus(db, doc.id, 'error', message);
    return { documentId: doc.id, status: 'error' };
  }
}
```

```ts
import { and, asc, count, eq, inArray, isNull } from 'drizzle-orm';
import type { Db } from '../db/client';
import {
  candidates,
  documents,
  knownWords,
  occurrences,
  words,
  type MeaningStatus,
} from '../db/schema';

export class ReviewError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'ReviewError';
  }
}

export interface SavedCards {
  occurrenceIds: number[];
  wordIds: number[];
}

// 검토 결과 저장: 선택 → 카드(words)+용례(occurrences), 선택 해제 → 아는 단어.
// 이미 카드인 단어를 해제하면 이번 용례만 건너뛰고 아는 단어로 넣지 않는다.
export function saveReviewSelection(
  db: Db,
  documentId: number,
  selectedCandidateIds: readonly number[],
): SavedCards {
  return db.transaction((tx) => {
    const doc = tx.select().from(documents).where(eq(documents.id, documentId)).get();
    if (!doc) throw new ReviewError('문서를 찾을 수 없습니다.');
    if (doc.status !== 'reviewing') throw new ReviewError('이미 저장한 문서입니다.');

    const selected = new Set(selectedCandidateIds);
    const list = tx.select().from(candidates).where(eq(candidates.documentId, documentId)).all();
    const saved: SavedCards = { occurrenceIds: [], wordIds: [] };

    for (const c of list) {
      if (!selected.has(c.id)) {
        if (!c.isExisting) {
          tx.insert(knownWords).values({ lemma: c.lemma }).onConflictDoNothing().run();
        }
        continue;
      }
      const word =
        tx.select().from(words).where(eq(words.lemma, c.lemma)).get() ??
        tx.insert(words).values({ lemma: c.lemma }).returning().get();
      const occurrence = tx
        .insert(occurrences)
        .values({ wordId: word.id, documentId, surface: c.surface, sentence: c.sentence })
        .returning({ id: occurrences.id })
        .get();
      saved.occurrenceIds.push(occurrence.id);
      saved.wordIds.push(word.id);
    }

    tx.delete(candidates).where(eq(candidates.documentId, documentId)).run();
    tx.update(documents)
      .set({ status: 'processing', errorMessage: null })
      .where(eq(documents.id, documentId))
      .run();
    return saved;
  });
}

export interface MeaningRequestRow {
  id: number;
  lemma: string;
  sentence: string;
}

export function listMeaningRequests(db: Db, occurrenceIds: readonly number[]): MeaningRequestRow[] {
  if (occurrenceIds.length === 0) return [];
  return db
    .select({ id: occurrences.id, lemma: words.lemma, sentence: occurrences.sentence })
    .from(occurrences)
    .innerJoin(words, eq(words.id, occurrences.wordId))
    .where(inArray(occurrences.id, [...occurrenceIds]))
    .orderBy(asc(occurrences.id))
    .all();
}

export function setMeanings(db: Db, results: { id: number; meaningKo: string }[]): void {
  db.transaction((tx) => {
    for (const r of results) {
      tx.update(occurrences)
        .set({ meaningKo: r.meaningKo, meaningStatus: 'done' })
        .where(eq(occurrences.id, r.id))
        .run();
    }
  });
}

export function markMeaningsFailed(db: Db, occurrenceIds: readonly number[]): void {
  if (occurrenceIds.length === 0) return;
  db.update(occurrences)
    .set({ meaningStatus: 'failed' })
    .where(inArray(occurrences.id, [...occurrenceIds]))
    .run();
}

export function listWordsMissingPhonetic(
  db: Db,
  wordIds: readonly number[],
): { id: number; lemma: string }[] {
  if (wordIds.length === 0) return [];
  return db
    .select({ id: words.id, lemma: words.lemma })
    .from(words)
    .where(and(inArray(words.id, [...new Set(wordIds)]), isNull(words.phonetic)))
    .all();
}

export function setPhonetic(db: Db, wordId: number, phonetic: string): void {
  db.update(words).set({ phonetic }).where(eq(words.id, wordId)).run();
}

export interface OccurrenceView {
  id: number;
  lemma: string;
  phonetic: string | null;
  surface: string;
  sentence: string;
  meaningKo: string | null;
  meaningStatus: MeaningStatus;
}

export function listOccurrencesByDocument(db: Db, documentId: number): OccurrenceView[] {
  return db
    .select({
      id: occurrences.id,
      lemma: words.lemma,
      phonetic: words.phonetic,
      surface: occurrences.surface,
      sentence: occurrences.sentence,
      meaningKo: occurrences.meaningKo,
      meaningStatus: occurrences.meaningStatus,
    })
    .from(occurrences)
    .innerJoin(words, eq(words.id, occurrences.wordId))
    .where(eq(occurrences.documentId, documentId))
    .orderBy(asc(words.lemma))
    .all();
}

export function countOccurrencesByDocument(db: Db): Map<number, number> {
  const rows = db
    .select({ documentId: occurrences.documentId, total: count() })
    .from(occurrences)
    .groupBy(occurrences.documentId)
    .all();
  return new Map(rows.map((r) => [r.documentId, r.total]));
}
```

```ts
// items를 최대 limit개씩 동시에 처리한다. 결과 순서는 입력 순서와 같다.
export async function mapWithConcurrency<T, R>(
  items: readonly T[],
  limit: number,
  fn: (item: T) => Promise<R>,
): Promise<R[]> {
  const results = new Array<R>(items.length);
  let next = 0;
  const worker = async () => {
    while (next < items.length) {
      const index = next++;
      results[index] = await fn(items[index]);
    }
  };
  await Promise.all(Array.from({ length: Math.min(limit, items.length) }, worker));
  return results;
}
```

`src/lib/phonetic/lookup.ts`:

```ts
const ENDPOINT = 'https://api.dictionaryapi.dev/api/v2/entries/en/';
const TIMEOUT_MS = 5000;

export type FetchLike = (
  url: string,
  init?: { signal?: AbortSignal },
) => Promise<{ ok: boolean; json(): Promise<unknown> }>;

interface Entry {
  phonetic?: unknown;
  phonetics?: { text?: unknown }[];
}

const nonEmpty = (value: unknown): value is string =>
  typeof value === 'string' && value.trim().length > 0;

function pickPhonetic(data: unknown): string | null {
  if (!Array.isArray(data)) return null;
  for (const entry of data as Entry[]) {
    if (nonEmpty(entry.phonetic)) return entry.phonetic.trim();
    for (const p of entry.phonetics ?? []) {
      if (nonEmpty(p.text)) return p.text.trim();
    }
  }
  return null;
}

// Free Dictionary API에서 발음기호를 찾는다. 404·오류·시간 초과는 null(재시도하지 않음).
export async function lookupPhonetic(word: string, fetchFn: FetchLike = fetch): Promise<string | null> {
  try {
    const res = await fetchFn(ENDPOINT + encodeURIComponent(word), {
      signal: AbortSignal.timeout(TIMEOUT_MS),
    });
    if (!res.ok) return null;
    return pickPhonetic(await res.json());
  } catch {
    return null;
  }
}
```

```ts
export interface MeaningRequest {
  id: number;
  word: string;
  sentence: string;
}

export interface MeaningResult {
  id: number;
  meaningKo: string;
}

// 단어 묶음 하나를 받아 문맥 뜻을 돌려주는 모델. 실제 구현은 claude.ts, 테스트는 가짜를 쓴다.
export type MeaningModel = (batch: MeaningRequest[]) => Promise<MeaningResult[]>;

export const MEANING_BATCH_SIZE = 50;

export interface GenerateOutcome {
  done: MeaningResult[];
  failedIds: number[];
}

// 50개씩 나눠 요청한다. 응답에 빠졌거나 빈 뜻, 실패한 묶음은 failedIds로 돌려준다.
export async function generateMeanings(
  requests: MeaningRequest[],
  model: MeaningModel,
): Promise<GenerateOutcome> {
  const outcome: GenerateOutcome = { done: [], failedIds: [] };
  for (let i = 0; i < requests.length; i += MEANING_BATCH_SIZE) {
    const batch = requests.slice(i, i + MEANING_BATCH_SIZE);
    const ids = new Set(batch.map((r) => r.id));
    let results: MeaningResult[] = [];
    try {
      results = await model(batch);
    } catch (err) {
      console.error('문맥 뜻 생성 실패', err);
    }
    const answered = new Map<number, string>();
    for (const r of results) {
      const meaning = r.meaningKo.trim();
      if (ids.has(r.id) && meaning.length > 0 && !answered.has(r.id)) answered.set(r.id, meaning);
    }
    for (const req of batch) {
      const meaning = answered.get(req.id);
      if (meaning === undefined) outcome.failedIds.push(req.id);
      else outcome.done.push({ id: req.id, meaningKo: meaning });
    }
  }
  return outcome;
}
```

```ts
import Anthropic from '@anthropic-ai/sdk';
import { zodOutputFormat } from '@anthropic-ai/sdk/helpers/zod';
import { z } from 'zod';
import type { MeaningModel } from './generate';

export const MEANING_MODEL_ID = 'claude-haiku-5-5';

const SYSTEM_PROMPT = [
  '너는 영어 논문·발표 자료를 읽는 한국인 학습자를 돕는다.',
  '각 항목의 word가 sentence 안에서 쓰인 의미를 한국어로 짧게(2~15자) 답한다.',
  '사전의 첫 뜻이 아니라 그 문장 속 의미를 쓰고, 품사에 맞게 쓴다(동사는 "~하다", 형용사는 "~한").',
  '모든 id에 하나씩 답한다.',
].join('\n');

const OutputSchema = z.object({
  meanings: z.array(z.object({ id: z.number().int(), meaning_ko: z.string() })),
});

// Claude Haiku 5.5로 문맥 뜻을 만든다. 키는 ANTHROPIC_API_KEY에서 읽는다(서버 전용).
export function createClaudeMeaningModel(client: Anthropic = new Anthropic()): MeaningModel {
  return async (batch) => {
    const response = await client.messages.parse({
      model: MEANING_MODEL_ID,
      max_tokens: 16000,
      system: SYSTEM_PROMPT,
      messages: [
        {
          role: 'user',
          content: JSON.stringify(batch.map((r) => ({ id: r.id, word: r.word, sentence: r.sentence }))),
        },
      ],
      output_config: { effort: 'low', format: zodOutputFormat(OutputSchema) },
    });
    if (response.stop_reason === 'refusal') throw new Error('모델이 요청을 거절했습니다.');
    if (!response.parsed_output) throw new Error(`응답을 해석하지 못했습니다: ${response.stop_reason}`);
    return response.parsed_output.meanings.map((m) => ({ id: m.id, meaningKo: m.meaning_ko }));
  };
}
```

```ts
import { mapWithConcurrency } from '@/lib/async/pool';
import { createClaudeMeaningModel } from '@/lib/meaning/claude';
import { generateMeanings, type MeaningModel } from '@/lib/meaning/generate';
import { lookupPhonetic } from '@/lib/phonetic/lookup';
import type { Db } from '../db/client';
import {
  listMeaningRequests,
  listWordsMissingPhonetic,
  markMeaningsFailed,
  setMeanings,
  setPhonetic,
  type SavedCards,
} from '../repo/cards';
import { updateDocumentStatus } from '../repo/documents';

const PHONETIC_CONCURRENCY = 5;

export interface EnrichDeps {
  // null이면 API 키가 없는 것: 뜻은 모두 failed로 둔다.
  meaningModel: MeaningModel | null;
  lookupPhonetic: (word: string) => Promise<string | null>;
}

export function defaultEnrichDeps(): EnrichDeps {
  return {
    meaningModel: process.env.ANTHROPIC_API_KEY ? createClaudeMeaningModel() : null,
    lookupPhonetic: (word) => lookupPhonetic(word),
  };
}

async function fillMeanings(db: Db, occurrenceIds: number[], model: MeaningModel | null) {
  const rows = listMeaningRequests(db, occurrenceIds);
  if (!model) {
    markMeaningsFailed(db, rows.map((r) => r.id));
    return;
  }
  const outcome = await generateMeanings(
    rows.map((r) => ({ id: r.id, word: r.lemma, sentence: r.sentence })),
    model,
  );
  setMeanings(db, outcome.done);
  markMeaningsFailed(db, outcome.failedIds);
}

async function fillPhonetics(db: Db, wordIds: number[], lookup: EnrichDeps['lookupPhonetic']) {
  const missing = listWordsMissingPhonetic(db, wordIds);
  await mapWithConcurrency(missing, PHONETIC_CONCURRENCY, async (word) => {
    const phonetic = await lookup(word.lemma);
    if (phonetic) setPhonetic(db, word.id, phonetic);
  });
}

// 저장 직후 문맥 뜻과 발음기호를 채우고 문서를 완료로 바꾼다. 개별 실패는 용례 상태로 남긴다.
export async function enrichDocument(
  db: Db,
  documentId: number,
  saved: SavedCards,
  deps: EnrichDeps,
): Promise<void> {
  await Promise.all([
    fillMeanings(db, saved.occurrenceIds, deps.meaningModel),
    fillPhonetics(db, saved.wordIds, deps.lookupPhonetic),
  ]);
  updateDocumentStatus(db, documentId, 'done');
}
```

```ts
import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

// 검토 결과를 저장하고, 뜻과 발음을 채운 뒤 문서 단어 목록으로 이동한다.
export async function saveReview(
  documentId: number,
  selectedCandidateIds: number[],
): Promise<{ error: string }> {
  const db = getDb();
  let saved: SavedCards;
  try {
    saved = saveReviewSelection(db, documentId, selectedCandidateIds);
  } catch (err) {
    if (err instanceof ReviewError) return { error: err.message };
    throw err;
  }

  await enrichDocument(db, documentId, saved, defaultEnrichDeps());
  revalidatePath("/documents");
  redirect(`/documents/${documentId}`);
```

```tsx
import Link from "next/link";
import { connection } from "next/server";
import { Suspense } from "react";
import { getDb } from "@/server/db/client";
import type { DocumentStatus } from "@/server/db/schema";
import { countOccurrencesByDocument } from "@/server/repo/cards";
import { countCandidatesByDocument } from "@/server/repo/candidates";
import { listDocuments } from "@/server/repo/documents";
import { UploadForm } from "./upload-form";

const STATUS_LABEL: Record<DocumentStatus, string> = {
  extracting: "추출 중",
  reviewing: "검토 대기",
  processing: "처리 중",
  done: "완료",
  error: "오류",
};

export default function DocumentsPage() {
  return (
    <main className="space-y-6">
      <h1 className="text-2xl font-semibold">문서</h1>
      <UploadForm />
      <Suspense fallback={<p>불러오는 중…</p>}>
        <DocumentList />
      </Suspense>
    </main>
  );
}

async function DocumentList() {
  await connection();
  const db = getDb();
  const docs = listDocuments(db);
  const candidateCounts = countCandidatesByDocument(db);
  const wordCounts = countOccurrencesByDocument(db);

  if (docs.length === 0) return <p className="text-gray-500">아직 올린 문서가 없습니다.</p>;

  return (
    <table className="w-full text-left text-sm">
      <thead>
        <tr className="border-b">
          <th className="py-2">제목</th>
          <th>종류</th>
          <th>상태</th>
          <th>단어 수</th>
          <th>올린 날짜</th>
        </tr>
      </thead>
      <tbody>
        {docs.map((doc) => (
          <tr key={doc.id} className="border-b">
            <td className="py-2">
              {doc.status === "reviewing" ? (
                <Link href={`/documents/${doc.id}/review`} className="underline">
                  {doc.title}
                </Link>
              ) : doc.status === "done" || doc.status === "processing" ? (
                <Link href={`/documents/${doc.id}`} className="underline">
                  {doc.title}
                </Link>
              ) : (
                doc.title
              )}
            </td>
            <td>{doc.kind === "paper" ? "논문" : "발표"}</td>
            <td>
              {STATUS_LABEL[doc.status]}
              {doc.errorMessage && (
                <span className="block text-xs text-red-600">{doc.errorMessage}</span>
              )}
            </td>
            <td>
              {doc.status === "reviewing"
                ? `후보 ${candidateCounts.get(doc.id) ?? 0}`
                : (wordCounts.get(doc.id) ?? 0)}
            </td>
            <td>{doc.createdAt.toLocaleDateString("ko-KR")}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}
```

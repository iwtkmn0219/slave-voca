```ts
function fakeFetch(outcomes: Outcome[]) {
  const called: string[] = [];
  let i = 0;
  const fetchFn: FetchLike = async (url) => {
    called.push(decodeURIComponent(url.split('/').pop()!));
    const outcome = outcomes[i++] ?? 'ok';
    if (outcome === 'timeout') throw new DOMException('timed out', 'TimeoutError');
    if (outcome === 'down') return { ok: false, status: 522, json: async () => ({}) };
    if (outcome === 'missing') return { ok: false, status: 404, json: async () => ({}) };
    return { ok: true, status: 200, json: async () => [{ phonetic: '/x/' }] };
  };
  return { fetchFn, called };
}

async function lookupAll(lookup: (w: string) => Promise<string | null>, count: number) {
  const results: (string | null)[] = [];
  for (let n = 0; n < count; n++) results.push(await lookup(`w${n}`));
  return results;
}

describe('createPhoneticLookup', () => {
  it('stops calling the API after 5 consecutive failures', async () => {
    const { fetchFn, called } = fakeFetch(['down', 'timeout', 'down', 'timeout', 'down']);
    const results = await lookupAll(createPhoneticLookup(fetchFn), 8);
    expect(called).toHaveLength(5);
    expect(results).toEqual(Array(8).fill(null));
  });

  it('does not trip on 404s, which mean the API is alive', async () => {
    const { fetchFn, called } = fakeFetch(Array(10).fill('missing'));
    await lookupAll(createPhoneticLookup(fetchFn), 11);
    expect(called).toHaveLength(11);

    let calls = 0;
```

- [ ] **Step 4: 429를 장애로 분류** — `src/lib/phonetic/lookup.ts` 전체

```ts
const ENDPOINT = 'https://api.dictionaryapi.dev/api/v2/entries/en/';
const TIMEOUT_MS = 5000;
// 사전 API가 응답하지 않으면 요청마다 5초를 기다리게 된다. 연속으로 이만큼 실패하면 남은 조회를 건너뛴다.
const MAX_CONSECUTIVE_FAILURES = 5;

export type FetchLike = (
  url: string,
  init?: { signal?: AbortSignal },
) => Promise<{ ok: boolean; status?: number; json(): Promise<unknown> }>;

// found: 발음기호 있음, missing: 사전에 없거나 발음기호가 없음(API는 정상), failed: 시간 초과·네트워크·5xx·429
type LookupOutcome =
  | { kind: 'found'; phonetic: string }
  | { kind: 'missing' }
  | { kind: 'failed' };

interface Entry {
  phonetic?: unknown;
  phonetics?: { text?: unknown }[];
}

const nonEmpty = (value: unknown): value is string =>
  typeof value === 'string' && value.trim().length > 0;

function pickPhonetic(data: unknown): string | null {
  if (!Array.isArray(data)) return null;
  for (const entry of data as Entry[]) {
    if (nonEmpty(entry.phonetic)) return entry.phonetic.trim();
    for (const p of entry.phonetics ?? []) {
      if (nonEmpty(p.text)) return p.text.trim();
    }
  }
  return null;
}

// 요청 한도 초과(429)와 5xx는 장애다. 404 같은 나머지 4xx는 "사전에 없음"으로 본다.
const isOutage = (status: number | undefined) =>
  status === undefined || status === 429 || status >= 500;

async function fetchPhonetic(word: string, fetchFn: FetchLike): Promise<LookupOutcome> {
  try {
    const res = await fetchFn(ENDPOINT + encodeURIComponent(word), {
      signal: AbortSignal.timeout(TIMEOUT_MS),
    });
    if (!res.ok) return isOutage(res.status) ? { kind: 'failed' } : { kind: 'missing' };
    const phonetic = pickPhonetic(await res.json());
    return phonetic ? { kind: 'found', phonetic } : { kind: 'missing' };
  } catch {
    return { kind: 'failed' };
  }
}

// Free Dictionary API에서 발음기호를 찾는다. 404·오류·시간 초과는 null(재시도하지 않음).
export async function lookupPhonetic(word: string, fetchFn: FetchLike = fetch): Promise<string | null> {
  const outcome = await fetchPhonetic(word, fetchFn);
  return outcome.kind === 'found' ? outcome.phonetic : null;
}

// 문서 하나를 채우는 동안 쓰는 조회 함수. 연속 5번 장애가 나면 이후 단어는 API를 부르지 않고 null을 준다.
// 404(사전에 없음)는 API가 살아 있다는 뜻이라 장애로 세지 않고, 한 번이라도 찾으면 연속 횟수를 0으로 되돌린다.
export function createPhoneticLookup(fetchFn: FetchLike = fetch): (word: string) => Promise<string | null> {
  let consecutiveFailures = 0;
  return async (word) => {
    if (consecutiveFailures >= MAX_CONSECUTIVE_FAILURES) return null;
    const outcome = await fetchPhonetic(word, fetchFn);
    consecutiveFailures = outcome.kind === 'failed' ? consecutiveFailures + 1 : 0;
    return outcome.kind === 'found' ? outcome.phonetic : null;
  };
}
```

```ts
// CMUdict의 ARPAbet 기호 → IPA. 모음은 강세 숫자(0·1·2)를 뗀 이름으로 찾는다.
const VOWELS: Record<string, string> = {
  AA: 'ɑ',
  AE: 'æ',
  AO: 'ɔ',
  AW: 'aʊ',
  AY: 'aɪ',
  EH: 'ɛ',
  EY: 'eɪ',
  IH: 'ɪ',
  IY: 'i',
  OW: 'oʊ',
  OY: 'ɔɪ',
  UH: 'ʊ',
  UW: 'u',
};

const CONSONANTS: Record<string, string> = {
  B: 'b',
  CH: 'tʃ',
  D: 'd',
  DH: 'ð',
  F: 'f',
  G: 'ɡ',
  HH: 'h',
  JH: 'dʒ',
  K: 'k',
  L: 'l',
  M: 'm',
  N: 'n',
  NG: 'ŋ',
  P: 'p',
  R: 'r',
  S: 's',
  SH: 'ʃ',
  T: 't',
  TH: 'θ',
  V: 'v',
  W: 'w',
  Y: 'j',
  Z: 'z',
  ZH: 'ʒ',
};

const STRESS_MARK: Record<string, string> = { '1': 'ˈ', '2': 'ˌ' };

interface Phone {
  ipa: string;
  vowel: boolean;
  stress: string; // '' | 'ˈ' | 'ˌ'
}

function toPhone(symbol: string): Phone | null {
  const match = /^([A-Z]+)([012])?$/.exec(symbol);
  if (!match) return null;
  const [, base, digit] = match;
  if (base === 'AH') return { ipa: digit === '0' ? 'ə' : 'ʌ', vowel: true, stress: STRESS_MARK[digit] ?? '' };
  if (base === 'ER') return { ipa: digit === '0' ? 'ɚ' : 'ɝ', vowel: true, stress: STRESS_MARK[digit] ?? '' };
  if (VOWELS[base]) return { ipa: VOWELS[base], vowel: true, stress: STRESS_MARK[digit] ?? '' };
  if (CONSONANTS[base] && digit === undefined) return { ipa: CONSONANTS[base], vowel: false, stress: '' };
  return null;
}

// "R OW0 B AH1 S T" → "/roʊˈbʌst/". 강세 기호는 강세 모음 바로 앞 자음 하나 앞에 붙인다(음절 분리 없는 근사).
// 모음이 하나뿐인 단어는 강세 기호를 생략한다. 모르는 기호가 있으면 null.
export function arpabetToIpa(arpabet: string): string | null {
  const phones = arpabet.trim().split(/\s+/).map(toPhone);
  if (phones.length === 0 || phones.some((p) => p === null)) return null;
  const list = phones as Phone[];
  const markStress = list.filter((p) => p.vowel).length > 1;

  const out: string[] = list.map((p) => p.ipa);
  for (let i = list.length - 1; i >= 0; i--) {
    const p = list[i];
    if (!markStress || !p.vowel || p.stress === '') continue;
    const at = i > 0 && !list[i - 1].vowel ? i - 1 : i;
    out[at] = p.stress + out[at];
  }
  return `/${out.join('')}/`;
}
```

`src/lib/phonetic/cmudict.ts`:

```ts
import fs from 'node:fs';
import path from 'node:path';
import { arpabetToIpa } from './arpabet';

export type PhoneticSource = (word: string) => string | null;

// "word PH ONES" 줄 목록으로 조회 함수를 만든다. 같은 단어가 여러 번 나오면 첫 줄을 쓴다.
export function createCmudict(text: string): PhoneticSource {
  const entries = new Map<string, string>();
  for (const line of text.split(/\r?\n/)) {
    const space = line.indexOf(' ');
    if (space <= 0) continue;
    const word = line.slice(0, space);
    if (!entries.has(word)) entries.set(word, line.slice(space + 1));
  }
  return (word) => {
    const arpabet = entries.get(word.toLowerCase());
    return arpabet === undefined ? null : arpabetToIpa(arpabet);
  };
}

let cached: PhoneticSource | undefined;

// data/cmudict.txt(약 13만 단어)를 처음 쓸 때 한 번만 읽는다. 번들에 넣지 않으려고 fs로 읽는다.
export function cmudictLookup(word: string): string | null {
  cached ??= createCmudict(fs.readFileSync(path.join(process.cwd(), 'data', 'cmudict.txt'), 'utf8'));
  return cached(word);
}

// API 조회가 실패하거나 못 찾으면 CMUdict로 채운다.
export function withCmudictFallback(
  lookup: (word: string) => Promise<string | null>,
  fallback: PhoneticSource = cmudictLookup,
): (word: string) => Promise<string | null> {
  return async (word) => (await lookup(word)) ?? fallback(word);
}
```

```ts
  const lines: string[] = [];
  for (const raw of dict.split(/\r?\n/)) {

  writeFileSync('data/cmudict-LICENSE', license);
  console.log(`data/cmudict.txt: ${lines.length}개 단어`);
}
```

```ts
import { fsrs, Rating, type Card, type Grade } from 'ts-fsrs';

export type Answer = 'again' | 'hard' | 'good';

export const ANSWERS: readonly Answer[] = ['again', 'hard', 'good'];

// 모름 → Again, 애매 → Hard, 앎 → Good (스펙 8.6)
export const ANSWER_GRADE: Record<Answer, Grade> = {
  again: Rating.Again,
  hard: Rating.Hard,
  good: Rating.Good,
};

export interface SrsState {
  due: Date;
  stability: number;
  difficulty: number;
  elapsedDays: number;
  scheduledDays: number;
  learningSteps: number;
  reps: number;
  lapses: number;
  state: number; // ts-fsrs State: 0 New, 1 Learning, 2 Review, 3 Relearning
  lastReview: Date | null;
}

// ts-fsrs 기본 파라미터. 퍼즈(복습일 무작위 흔들기)는 꺼서 같은 응답이면 같은 일정이 나오게 한다.
const scheduler = fsrs({ enable_fuzz: false });

function toCard(s: SrsState): Card {
  return {
    due: s.due,
    stability: s.stability,
    difficulty: s.difficulty,
    elapsed_days: s.elapsedDays,
    scheduled_days: s.scheduledDays,
    learning_steps: s.learningSteps,
    reps: s.reps,
    lapses: s.lapses,
    state: s.state,
    last_review: s.lastReview ?? undefined,
  };
}

export function applyAnswer(current: SrsState, answer: Answer, now: Date): SrsState {
  const next = scheduler.next(toCard(current), now, ANSWER_GRADE[answer]).card;
  return {
    due: next.due,
    stability: next.stability,
    difficulty: next.difficulty,
    elapsedDays: next.elapsed_days,
    scheduledDays: next.scheduled_days,
    learningSteps: next.learning_steps,
    reps: next.reps,
    lapses: next.lapses,
    state: next.state,
    lastReview: next.last_review ?? now,
  };
}

// "오늘"은 서버 로컬 시간 자정부터다.
export function startOfLocalDay(now: Date): Date {
  const day = new Date(now);
  day.setHours(0, 0, 0, 0);
  return day;
}

export function newCardSlots(dailyNewLimit: number, introducedToday: number): number {
  return Math.max(0, dailyNewLimit - introducedToday);
}
```

```ts
import { and, asc, count, desc, eq, gte, inArray, lte, ne } from 'drizzle-orm';
import {
  ANSWER_GRADE,
  applyAnswer,
  newCardSlots,
  startOfLocalDay,
  type Answer,
} from '@/lib/srs/scheduler';
import type { Db } from '../db/client';
import { documents, occurrences, reviewLogs, words, type MeaningStatus, type WordRow } from '../db/schema';

const NEW_STATE = 0;
const USAGES_PER_CARD = 2;

export class StudyError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'StudyError';
  }
}

export interface Usage {
  surface: string;
  sentence: string;
  meaningKo: string | null;
  meaningStatus: MeaningStatus;
  documentTitle: string;
}

export interface StudyCard {
  wordId: number;
  lemma: string;
  phonetic: string | null;
  isNew: boolean;
  usages: Usage[]; // 최신 용례부터 최대 2개
}

function countIntroducedToday(db: Db, now: Date): number {
  const row = db
    .select({ total: count() })
    .from(words)
    .where(gte(words.introducedAt, startOfLocalDay(now)))
    .get();
  return row?.total ?? 0;
}

function dueReviews(db: Db, now: Date): WordRow[] {
  return db
    .select()
    .from(words)
    .where(and(ne(words.state, NEW_STATE), lte(words.due, now)))
    .orderBy(asc(words.due), asc(words.id))
    .all();
}

function newCards(db: Db, limit: number): WordRow[] {
  if (limit <= 0) return [];
  return db
    .select()
    .from(words)
    .where(eq(words.state, NEW_STATE))
    .orderBy(asc(words.createdAt), asc(words.id))
    .limit(limit)
    .all();
}

function usagesByWord(db: Db, wordIds: number[]): Map<number, Usage[]> {
  const byWord = new Map<number, Usage[]>();
  if (wordIds.length === 0) return byWord;
  const rows = db
    .select({
      wordId: occurrences.wordId,
      surface: occurrences.surface,
      sentence: occurrences.sentence,
      meaningKo: occurrences.meaningKo,
      meaningStatus: occurrences.meaningStatus,
      documentTitle: documents.title,
    })
    .from(occurrences)
    .innerJoin(documents, eq(documents.id, occurrences.documentId))
    .where(inArray(occurrences.wordId, wordIds))
    .orderBy(desc(occurrences.createdAt), desc(occurrences.id))
    .all();
  for (const { wordId, ...usage } of rows) {
    const list = byWord.get(wordId) ?? [];
    if (list.length < USAGES_PER_CARD) list.push(usage);
    byWord.set(wordId, list);
  }
  return byWord;
}

// 오늘의 큐: 복습할 때가 된 카드(due 순) + 새 카드(만든 순, 하루 상한까지). 스펙 8.6.
export function getStudyQueue(db: Db, now: Date, dailyNewLimit: number): StudyCard[] {
  const slots = newCardSlots(dailyNewLimit, countIntroducedToday(db, now));
  const picked = [...dueReviews(db, now), ...newCards(db, slots)];
  const usages = usagesByWord(db, picked.map((w) => w.id));
  return picked.map((w) => ({
    wordId: w.id,
    lemma: w.lemma,
    phonetic: w.phonetic,
    isNew: w.state === NEW_STATE,
    usages: usages.get(w.id) ?? [],
  }));
}

export function countStudyToday(
  db: Db,
  now: Date,
  dailyNewLimit: number,
): { review: number; fresh: number } {
  const review =
    db
      .select({ total: count() })
      .from(words)
      .where(and(ne(words.state, NEW_STATE), lte(words.due, now)))
      .get()?.total ?? 0;
  const waiting =
    db.select({ total: count() }).from(words).where(eq(words.state, NEW_STATE)).get()?.total ?? 0;
  const slots = newCardSlots(dailyNewLimit, countIntroducedToday(db, now));
  return { review, fresh: Math.min(waiting, slots) };
}

// 응답을 기록하고 다음 복습 일정을 저장한다. 새 카드는 처음 응답한 시각을 introducedAt으로 남긴다.
export function recordAnswer(db: Db, wordId: number, answer: Answer, now: Date): { due: Date } {
  return db.transaction((tx) => {
    const word = tx.select().from(words).where(eq(words.id, wordId)).get();
    if (!word) throw new StudyError('카드를 찾을 수 없습니다.');
    const next = applyAnswer(word, answer, now);
    tx.update(words)
      .set({ ...next, introducedAt: word.introducedAt ?? now })
      .where(eq(words.id, wordId))
      .run();
    tx.insert(reviewLogs).values({ wordId, rating: ANSWER_GRADE[answer], reviewedAt: now }).run();
    return { due: next.due };
  });
}
```

```ts
export function advanceQueue<T>(queue: readonly T[], answer: Answer): T[] {
  if (queue.length === 0) return [];
  const [head, ...rest] = queue;
  return answer === 'again' ? [...rest, head] : rest;
```

```ts
"use server";

import { ANSWERS, type Answer } from "@/lib/srs/scheduler";
import { getDb } from "@/server/db/client";
import { recordAnswer, StudyError } from "@/server/repo/study";

// 복습 응답 하나를 저장한다. 실패하면 화면에 보여줄 메시지를 돌려준다.
export async function answerCard(wordId: number, answer: Answer): Promise<{ error: string | null }> {
  if (!Number.isInteger(wordId) || !ANSWERS.includes(answer)) {
    return { error: "잘못된 요청입니다." };
  }
  try {
    recordAnswer(getDb(), wordId, answer, new Date());
    return { error: null };
  } catch (err) {
    if (err instanceof StudyError) return { error: err.message };
    throw err;
  }
}
```

```tsx
  { answer: "hard", label: "애매", className: "bg-yellow-600" },
  { answer: "good", label: "앎", className: "bg-green-700" },
];

export function StudySession({ cards }: { cards: StudyCard[] }) {
  const [queue, setQueue] = useState(cards);
  const [revealed, setRevealed] = useState(false);
  const [answered, setAnswered] = useState(0);
  const [error, setError] = useState<string | null>(null);
  const [saving, startSaving] = useTransition();

  if (queue.length === 0) {
    return (
      <div className="space-y-3">
        <p className="text-lg">오늘 복습을 마쳤습니다. {answered > 0 && `(응답 ${answered}번)`}</p>
        <Link href="/" className="underline">
          홈으로
        </Link>
      </div>
    );
  }

  const card = queue[0];
  const answer = (choice: Answer) =>
    startSaving(async () => {
      const result = await answerCard(card.wordId, choice);
      if (result.error) {
        setError(result.error);
        return;
      }
      setError(null);
      setAnswered((n) => n + 1);
      setQueue((q) => advanceQueue(q, choice));
      setRevealed(false);
    });

  return (
    <div className="space-y-6">
      <p className="text-sm text-gray-500">남은 카드 {queue.length}장</p>

        <div className="flex items-center gap-3">
          <h2 className="text-3xl font-semibold">{card.lemma}</h2>
          <SpeakButton text={card.lemma} />
          {card.isNew && <span className="rounded bg-blue-100 px-2 text-xs">새 카드</span>}
        </div>

        {revealed ? (
          <div className="space-y-4 pt-2">
            <p className="text-gray-600">{card.phonetic ?? "발음기호 없음"}</p>

                <p className="text-sm">
                  {highlight(u.sentence, u.surface).map((seg, j) =>
                    seg.match ? <mark key={j}>{seg.text}</mark> : <span key={j}>{seg.text}</span>,
                  )}
                </p>
                <p className="text-xs text-gray-500">{u.documentTitle}</p>
              </div>

          {error}

      )}

  );
}
```
